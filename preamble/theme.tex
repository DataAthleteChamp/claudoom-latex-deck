\usetheme{metropolis}

\setbeamercolor{date}{fg=muted}

\setbeamercolor{footnote mark}{fg=accent}
\setbeamercolor{itemize item}{fg=accent}

\setbeamercolor{block title}{fg=accent, bg=slate}
